#include <stdio.h>
#include <string.h>

int main() {
    char str;
    int visited = {0};

    printf("Enter a string: ");
    
    if (fgets(str, sizeof(str), stdin) == NULL) {
        return 0;
    }

    str[strcspn(str, "\n")] = '\0';
    int len = strlen(str);
    int writeIndex = 0;

    for (int i = 0; i < len; i++) {
        unsigned char ch = str[i];
        if (visited[ch] == 0) {
            visited[ch] = 1;
            str[writeIndex] = str[i];
            writeIndex++;
        }
    }
    str[writeIndex] = '\0';

    printf("%s\n", str);

    return 0;
}
