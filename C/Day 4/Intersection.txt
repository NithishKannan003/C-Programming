#include <stdio.h>

int main() {
    int n, m;
    scanf("%d", &n);

    int a[n];
    for (int i = 0; i < n; i++)
        scanf("%d", &a[i]);

    scanf("%d", &m);

    int b[m];
    for (int i = 0; i < m; i++)
        scanf("%d", &b[i]);

    for (int i = 0; i < n; i++) {
        int found = 0;

        for (int j = 0; j < m; j++) {
            if (a[i] == b[j]) {
                found = 1;
                break;
            }
        }

        if (found) {
            int duplicate = 0;

            for (int k = 0; k < i; k++) {
                if (a[k] == a[i]) {
                    duplicate = 1;
                    break;
                }
            }

            if (!duplicate)
                printf("%d ", a[i]);
        }
    }

    return 0;
}
