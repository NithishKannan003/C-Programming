#include <stdio.h>

struct Book {
    char title[100];
    char author[100];
    float price;
};

int main() {
    struct Book b;

    scanf(" %[^\n]", b.title);
    scanf(" %[^\n]", b.author);
    scanf("%f", &b.price);

    printf("Title: %s\n", b.title);
    printf("Author: %s\n", b.author);
    printf("Price: %.2f\n", b.price);

    return 0;
}
