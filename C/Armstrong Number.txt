#include <stdio.h>

int main() {
    int n;
    printf("Enter a number: ");
    scanf("%d", &n);
    int dup = n;
    int c = 0;
    while(dup != 0) {
        c++;
        dup = dup / 10;
    }
    int n1 = n;
    int rem;
    int sum = 0;
    while(n != 0){
        int d = 1;
        rem = n % 10;
        for(int i = 0; i < c; i++){
            d = d * rem;
        }
        sum = sum + d;
        n = n / 10;
    }
    if(sum == n1){
        printf("Armstrong Number");
    }
    else{
        printf("Not Armstrong Number");
    }
    return 0;
}
